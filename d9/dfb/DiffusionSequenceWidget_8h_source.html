50 void dropEvent(QDropEvent* event) {
51 const QMimeData* mimeData = event->mimeData();
52 if (mimeData->hasUrls()) {
53 const QList<QUrl>& urls = mimeData->urls();
54 for (QList<QUrl>::ConstIterator i = urls.begin();i!=urls.end();++i) {
55 const QString& filename = i->toLocalFile();
56 bool valid = false;
57 emit newItem(filename,valid);
58 if (valid) {
59 const QFileInfo& info = QFileInfo(filename);
60 const unsigned nrows = rowCount();
61 setRowCount(nrows+1);
62 setRowHeight(nrows,15);
63 QTableWidgetItem* item0 = new QTableWidgetItem(info.baseName());
64 QTableWidgetItem* item1 = new QTableWidgetItem("0, 0, 0");
65 item0->setData(Qt::UserRole,QVariant(filename));
66 setItem(nrows,0,item0);
67 setItem(nrows,1,item1);
68 }
69 }
70 }
