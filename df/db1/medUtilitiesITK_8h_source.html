156 template <typename ImageType> static void updateMetadata(medAbstractData* outputData)
157 {
158 typename ImageType::Pointer outputImage = static_cast<ImageType*>(outputData->data());
159
160 outputData->setMetaData(medMetaDataKeys::Columns.key(),
161 QString::number(outputImage->GetLargestPossibleRegion().GetSize()[0]));
162 outputData->setMetaData(medMetaDataKeys::Rows.key(),
163 QString::number(outputImage->GetLargestPossibleRegion().GetSize()[1]));
164 outputData->setMetaData(medMetaDataKeys::Size.key(),
165 QString::number(outputImage->GetLargestPossibleRegion().GetSize()[2]));
166 outputData->setMetaData(medMetaDataKeys::SliceThickness.key(),
167 QString::number(outputImage->GetSpacing()[2]));
168
169 outputData->setMetaData(medMetaDataKeys::Orientation.key(),
170 QString::number(outputImage->GetDirection()[0][0]) +
171 QString(" ") +
172 QString::number(outputImage->GetDirection()[0][1]) +
173 QString(" ") +
174 QString::number(outputImage->GetDirection()[0][2]) +
175 QString(" ") +
176 QString::number(outputImage->GetDirection()[1][0]) +
177 QString(" ") +
178 QString::number(outputImage->GetDirection()[1][1]) +
179 QString(" ") +
180 QString::number(outputImage->GetDirection()[1][2]));
181

183 QString::number(outputImage->GetOrigin()[0]) +
184 QString(" ") +
185 QString::number(outputImage->GetOrigin()[1]) +
186 QString(" ") +
187 QString::number(outputImage->GetOrigin()[2]));

192 template <typename TargetClass, typename ImageType, typename... OtherArguments>
193 static bool callPixelSpecificFunction(int (TargetClass::*function)(medAbstractData*, OtherArguments...),
194 TargetClass* targetInstance, int* result, medAbstractData* inputData, OtherArguments... otherArgs)
195 {
196 if ((function != nullptr) && (inputData->identifier().startsWith("itkDataImage")))
197 {
198 ImageType* image = dynamic_cast<ImageType*>(static_cast<itk::ImageBase<ImageType::ImageDimension>*>(inputData->data()));
199
200 if (image)
201 {
202 *result = (targetInstance->*function)(inputData, otherArgs...);
203 return true;
204 }
205 }
206
207 return false;
208 }
